}

// The native reset establishes only the selected current value, not absence or
// contents of default nodes. This bisimulation proves observable independence; keys outside this view are not compared.
pub open spec fn same_view(a:World,z:World)->bool {
 a.cfg.conf.contains(key()) && z.cfg.conf.contains(key()) && setting_value(a)==setting_value(z) && a.calls==z.calls
}
pub open spec fn same_output(a:Out,z:Out)->bool {
 same_view(a.world,z.world) && a.error==z.error && a.listed==z.listed && a.attempted==z.attempted
}
pub proof fn shell_same_view(a:World,z:World,c:Text)
 requires same_view(a,z)
 ensures same_view(shell(a,c).world,shell(z,c).world),shell(a,c).error==shell(z,c).error
{}
pub proof fn execute_same_view(a:World,z:World,cs:Seq<Text>,i:int,dry:bool)
 requires same_view(a,z),0<=i<=cs.len()
 ensures same_output(execute(a,cs,i,dry),execute(z,cs,i,dry))
 decreases cs.len()-i
{
 if i<cs.len(){
  if !dry{shell_same_view(a,z,cs[i]);}
  let sa=if dry{Step{world:a,error:None}}else{shell(a,cs[i])};
  let sz=if dry{Step{world:z,error:None}}else{shell(z,cs[i])};
  if sa.error.is_none(){execute_same_view(sa.world,sz.world,cs,i+1,dry);}
 }
}
pub proof fn arbitrary_default(d:Option<nat>,file:File,dry:bool)
 ensures same_output(run(initial_with_default(d),file,dry),run(initial(),file,dry)),
 raw_value(run(initial_with_default(d),file,dry).world)==raw_value(run(initial(),file,dry).world)
{
 let p=prepare(file);
 if p.error.is_none(){execute_same_view(initial_with_default(d),initial(),p.commands,0,dry);}
}
